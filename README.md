-- Hoja "camas"

    ON UPDATE CASCADE ON DELETE RESTRICT

-- Vistas útiles
CREATE VIEW v_camas_detalle AS
SELECT c.id_cama, s.tipo_cama, s.piso, s.sala, c.estado,
       c.horas_ocupada, c.edad_paciente, c.espera_asignacion_min
FROM camas c JOIN salas s USING (id_sala);

CREATE VIEW v_ocupacion_actual AS
SELECT s.id_sala, s.tipo_cama, s.piso, s.sala,
       COUNT(*)                                   AS total_camas,
       SUM(c.estado IN ('Ocupada','Crítica'))     AS ocupadas,
       SUM(c.estado = 'Crítica')                  AS criticas,
       SUM(c.estado = 'Libre')                    AS libres,
       SUM(c.estado = 'En limpieza')              AS en_limpieza,
       SUM(c.estado = 'Reservada')                AS reservadas,
       ROUND(100*SUM(c.estado IN ('Ocupada','Crítica'))/COUNT(*),1) AS tasa_ocupacion_pct,
       ROUND(AVG(c.espera_asignacion_min),1)      AS espera_prom_min
FROM salas s JOIN camas c USING (id_sala)
GROUP BY s.id_sala, s.tipo_cama, s.piso, s.sala;
